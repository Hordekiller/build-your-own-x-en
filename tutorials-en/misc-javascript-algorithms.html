Core idea

Quick sort picks a pivot and partitions the array into smaller and larger halves, recursively. Every topic file pairs the implementation with its theoretical explanation and tests.

JavaScript// Binary search: O(log n)
function binarySearch(sortedArray, seekElement) {
    let start = 0;
    let end = sortedArray.length - 1;

    while (start <= end) {
        const middle = start + Math.floor((end - start) / 2);
        if (sortedArray[middle] === seekElement) return middle;
        if (sortedArray[middle] < seekElement) start = middle + 1;
        else end = middle - 1;
    }
    return -1;
}
