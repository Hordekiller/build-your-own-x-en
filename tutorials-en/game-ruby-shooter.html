Core idea

A Game class extends Gosu::Window: update handles input and logic every frame, draw renders; Literally the whole skeleton of a Ruby game.

Rubyrequire 'gosu'

class Game < Gosu::Window
  def initialize(width = 800, height = 600)
    super(width, height)
    self.caption = 'My Ruby Game'
    @player_x, @player_y = 400, 300
  end

  # per-frame logic update
  def update
    @player_x -= 5 if Gosu.button_down?(Gosu::KB_LEFT)
    @player_x += 5 if Gosu.button_down?(Gosu::KB_RIGHT)
    @player_y -= 5 if Gosu.button_down?(Gosu::KB_UP)
    @player_y += 5 if Gosu.button_down?(Gosu::KB_DOWN)
  end

  # per-frame render
  def draw
    # draw with Gosu::Image or simple shapes
  end
end
